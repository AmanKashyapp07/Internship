## Problem 1: Thread-Safe Bounded Queue (Classic Producer-Consumer)

**Task**: Implement a thread-safe queue with a maximum capacity. `push()` blocks when full; `pop()` blocks when empty.

```cpp
#include <iostream>
#include <queue>
#include <mutex>
#include <condition_variable>
#include <thread>

template<typename T>
class BoundedQueue {
    std::queue<T> q;
    std::mutex mtx;
    std::condition_variable not_full;   // signaled when an item is removed
    std::condition_variable not_empty;  // signaled when an item is added
    const size_t max_size;

public:
    explicit BoundedQueue(size_t capacity) : max_size(capacity) {}

    void push(T item) {
        std::unique_lock<std::mutex> lock(mtx);
        not_full.wait(lock, [this] { return q.size() < max_size; }); // block if full
        q.push(std::move(item));
        not_empty.notify_one(); // wake up one consumer
    }

    T pop() {
        std::unique_lock<std::mutex> lock(mtx);
        not_empty.wait(lock, [this] { return !q.empty(); }); // block if empty
        T item = std::move(q.front());
        q.pop();
        not_full.notify_one(); // wake up one producer
        return item;
    }

    size_t size() const {
        std::lock_guard<std::mutex> lock(mtx);
        return q.size();
    }
};

// --- Demo ---
int main() {
    BoundedQueue<int> bq(5);

    auto produce = [&]() {
        for (int i = 0; i < 10; ++i) {
            bq.push(i);
            std::cout << "[P] Pushed: " << i << "\n";
        }
    };

    auto consume = [&]() {
        for (int i = 0; i < 10; ++i) {
            int val = bq.pop();
            std::cout << "[C] Popped: " << val << "\n";
        }
    };

    std::thread producer(produce);
    std::thread consumer(consume);
    producer.join();
    consumer.join();
}
```

**Key concepts demonstrated**:
- Two condition variables for two different conditions (full vs. empty).
- Always use `while`/lambda predicate in `wait()` to handle spurious wakeups.
- `notify_one()` is sufficient when only one type of thread needs to wake up.

## Problem 2: Read-Write Lock (Readers-Writers)

**Task**: Implement a custom RW lock where multiple readers can read simultaneously, but a writer needs exclusive access.

```cpp
#include <mutex>
#include <condition_variable>
#include <thread>
#include <iostream>

class ReadWriteLock {
    std::mutex mtx;
    std::condition_variable cv;
    int readers = 0;       // count of active readers
    bool writing = false;  // is a writer active?

public:
    // Called by readers before reading
    void read_lock() {
        std::unique_lock<std::mutex> lock(mtx);
        cv.wait(lock, [this] { return !writing; }); // wait if a writer is active
        ++readers;
    }

    // Called by readers after reading
    void read_unlock() {
        std::unique_lock<std::mutex> lock(mtx);
        --readers;
        if (readers == 0)
            cv.notify_all(); // wake up waiting writers
    }

    // Called by writers before writing
    void write_lock() {
        std::unique_lock<std::mutex> lock(mtx);
        cv.wait(lock, [this] { return !writing && readers == 0; }); // wait for all readers AND writers
        writing = true;
    }

    // Called by writers after writing
    void write_unlock() {
        std::unique_lock<std::mutex> lock(mtx);
        writing = false;
        cv.notify_all(); // wake up all waiting readers and writers
    }
};

// RAII wrappers for convenience
struct ReadGuard {
    ReadWriteLock& rwl;
    ReadGuard(ReadWriteLock& r) : rwl(r) { rwl.read_lock(); }
    ~ReadGuard() { rwl.read_unlock(); }
};

struct WriteGuard {
    ReadWriteLock& rwl;
    WriteGuard(ReadWriteLock& r) : rwl(r) { rwl.write_lock(); }
    ~WriteGuard() { rwl.write_unlock(); }
};

// --- Demo ---
ReadWriteLock rwl;
int shared_data = 0;

void reader(int id) {
    ReadGuard rg(rwl);
    std::cout << "[Reader " << id << "] reads: " << shared_data << "\n";
}

void writer(int id) {
    WriteGuard wg(rwl);
    ++shared_data;
    std::cout << "[Writer " << id << "] wrote: " << shared_data << "\n";
}

int main() {
    std::thread r1(reader, 1), r2(reader, 2);
    std::thread w1(writer, 1);
    std::thread r3(reader, 3);
    r1.join(); r2.join(); w1.join(); r3.join();
}
```

> **Interview note**: This is a readers-preference implementation (writers can starve). A writer-preference version would have writers block new readers from entering when a writer is waiting.

## Problem 3: Thread-Safe LRU Cache (O(1) get & put)

**Task**: Implement an LRU Cache that is thread-safe. Both `get` and `put` must be O(1).

```cpp
#include <unordered_map>
#include <list>
#include <mutex>
#include <optional>
#include <iostream>

template<typename K, typename V>
class LRUCache {
    using ListIt = typename std::list<std::pair<K, V>>::iterator;

    int capacity;
    std::list<std::pair<K, V>> cache; // front = most recently used
    std::unordered_map<K, ListIt> index;
    mutable std::mutex mtx;

public:
    explicit LRUCache(int cap) : capacity(cap) {}

    std::optional<V> get(const K& key) {
        std::lock_guard<std::mutex> lock(mtx);
        auto it = index.find(key);
        if (it == index.end()) return std::nullopt; // cache miss
        // Move to front (mark as most recently used)
        cache.splice(cache.begin(), cache, it->second);
        return it->second->second;
    }

    void put(const K& key, const V& value) {
        std::lock_guard<std::mutex> lock(mtx);
        auto it = index.find(key);
        if (it != index.end()) {
            it->second->second = value;
            cache.splice(cache.begin(), cache, it->second);
            return;
        }
        if ((int)cache.size() == capacity) {
            // Evict least recently used (back of list)
            index.erase(cache.back().first);
            cache.pop_back();
        }
        cache.push_front({key, value});
        index[key] = cache.begin();
    }

    int size() const {
        std::lock_guard<std::mutex> lock(mtx);
        return (int)cache.size();
    }
};

int main() {
    LRUCache<int, std::string> lru(3);
    lru.put(1, "one");
    lru.put(2, "two");
    lru.put(3, "three");
    auto v = lru.get(1);
    std::cout << "get(1): " << (v ? *v : "miss") << "\n";  // one
    lru.put(4, "four");  // evicts 2 (LRU)
    auto v2 = lru.get(2);
    std::cout << "get(2): " << (v2 ? *v2 : "miss") << "\n"; // miss
}
```

## Problem 4: Thread-Safe Singleton (Double-Checked Locking)

**Task**: Implement a Singleton that is initialized only once, even in a multithreaded environment.

```cpp
#include <mutex>
#include <memory>
#include <iostream>

class Singleton {
    Singleton() { std::cout << "Singleton created\n"; }

public:
    Singleton(const Singleton&) = delete;
    Singleton& operator=(const Singleton&) = delete;

    // Method 1: Using static local variable (C++11 guaranteed thread-safe)
    // The best modern approach — no mutex needed!
    static Singleton& instance() {
        static Singleton obj; // Initialized exactly once, thread-safe since C++11
        return obj;
    }

    void doWork() { std::cout << "Singleton::doWork()\n"; }
};

// Method 2: Double-Checked Locking (older pattern, still asked in interviews)
class SingletonDCL {
    static SingletonDCL* instance_;
    static std::mutex mtx_;

    SingletonDCL() {}

public:
    static SingletonDCL* getInstance() {
        if (instance_ == nullptr) {          // First check (no lock — fast path)
            std::lock_guard<std::mutex> lock(mtx_);
            if (instance_ == nullptr) {      // Second check (with lock — avoid race)
                instance_ = new SingletonDCL();
            }
        }
        return instance_;
    }
};
SingletonDCL* SingletonDCL::instance_ = nullptr;
std::mutex SingletonDCL::mtx_;

int main() {
    auto& s1 = Singleton::instance();
    auto& s2 = Singleton::instance();
    s1.doWork();
    std::cout << "Same instance? " << (&s1 == &s2 ? "Yes" : "No") << "\n";
}
```

> **Interview tip**: Always prefer Method 1 (static local variable) in modern C++. Method 2 (DCL) is shown for historical awareness — interviewers sometimes ask you to explain it.

## Problem 5: Semaphore Implementation Using Mutex + CV

**Task**: Implement a counting semaphore from scratch using a mutex and condition variable.

```cpp
#include <mutex>
#include <condition_variable>
#include <thread>
#include <iostream>

class Semaphore {
    std::mutex mtx;
    std::condition_variable cv;
    int count;

public:
    explicit Semaphore(int initial_count) : count(initial_count) {}

    void wait() {  // P() / down() / acquire()
        std::unique_lock<std::mutex> lock(mtx);
        cv.wait(lock, [this] { return count > 0; }); // block if count == 0
        --count;
    }

    void signal() { // V() / up() / release()
        std::unique_lock<std::mutex> lock(mtx);
        ++count;
        cv.notify_one(); // wake up one waiting thread
    }
};

// --- Demo: Limit concurrent access to 3 threads at a time ---
Semaphore sem(3); // max 3 concurrent

void task(int id) {
    sem.wait();  // acquire
    std::cout << "Thread " << id << " entering critical section\n";
    std::this_thread::sleep_for(std::chrono::milliseconds(100));
    std::cout << "Thread " << id << " leaving critical section\n";
    sem.signal(); // release
}

int main() {
    std::thread threads[7];
    for (int i = 0; i < 7; ++i)
        threads[i] = std::thread(task, i);
    for (auto& t : threads)
        t.join();
}
```

## Problem 6: Barrier Synchronization

**Task**: Implement a barrier — all N threads must reach the barrier before any can proceed.

```cpp
#include <mutex>
#include <condition_variable>
#include <thread>
#include <iostream>

class Barrier {
    std::mutex mtx;
    std::condition_variable cv;
    int count;       // threads that have arrived
    int total;       // total threads expected
    int generation;  // prevents reuse issues

public:
    explicit Barrier(int n) : count(0), total(n), generation(0) {}

    void wait() {
        std::unique_lock<std::mutex> lock(mtx);
        int gen = generation;
        ++count;
        if (count == total) {
            // Last thread to arrive — release everyone
            ++generation;
            count = 0;
            cv.notify_all();
        } else {
            // Wait until the generation advances (barrier lifted)
            cv.wait(lock, [this, gen] { return generation != gen; });
        }
    }
};

Barrier barrier(5);

void phase_work(int id, const std::string& phase) {
    std::cout << "Thread " << id << " doing " << phase << "\n";
    std::this_thread::sleep_for(std::chrono::milliseconds(id * 10));
}

void worker(int id) {
    phase_work(id, "Phase 1");
    barrier.wait();  // Wait for all to finish Phase 1
    std::cout << "Thread " << id << " starting Phase 2\n";
    phase_work(id, "Phase 2");
}

int main() {
    std::thread threads[5];
    for (int i = 0; i < 5; ++i)
        threads[i] = std::thread(worker, i);
    for (auto& t : threads)
        t.join();
}
```

## Problem 7: Dining Philosophers (Deadlock-Free)

```cpp
#include <mutex>
#include <thread>
#include <iostream>
#include <array>
#include <chrono>

const int N = 5;
std::array<std::mutex, N> chopsticks;

void philosopher(int id) {
    int left  = id;
    int right = (id + 1) % N;

    for (int i = 0; i < 3; ++i) {
        // Think
        std::cout << "Philosopher " << id << " thinking\n";
        std::this_thread::sleep_for(std::chrono::milliseconds(10));

        // Odd-Even rule: break circular wait
        if (id % 2 == 0) {
            std::lock(chopsticks[left], chopsticks[right]); // atomic dual lock
        } else {
            std::lock(chopsticks[right], chopsticks[left]);
        }

        // RAII adoption of already-locked mutexes
        std::lock_guard<std::mutex> lg1(chopsticks[left],  std::adopt_lock);
        std::lock_guard<std::mutex> lg2(chopsticks[right], std::adopt_lock);

        // Eat
        std::cout << "Philosopher " << id << " eating\n";
        std::this_thread::sleep_for(std::chrono::milliseconds(10));
    } // chopsticks released here by RAII
}

int main() {
    std::array<std::thread, N> philosophers;
    for (int i = 0; i < N; ++i)
        philosophers[i] = std::thread(philosopher, i);
    for (auto& p : philosophers)
        p.join();
    std::cout << "All philosophers done\n";
}
```

> **Key trick**: `std::lock()` acquires **multiple mutexes simultaneously** using a deadlock-avoidance algorithm — never partial acquisition. This is the cleanest C++ solution.

## Problem 8: Thread Pool (Bonus — Systems Design)

**Task**: Implement a basic thread pool that processes tasks from a queue.

```cpp
#include <vector>
#include <thread>
#include <queue>
#include <functional>
#include <mutex>
#include <condition_variable>
#include <iostream>

class ThreadPool {
    std::vector<std::thread> workers;
    std::queue<std::function<void()>> tasks;
    std::mutex mtx;
    std::condition_variable cv;
    bool stop = false;

public:
    explicit ThreadPool(size_t num_threads) {
        for (size_t i = 0; i < num_threads; ++i) {
            workers.emplace_back([this] {
                while (true) {
                    std::function<void()> task;
                    {
                        std::unique_lock<std::mutex> lock(mtx);
                        cv.wait(lock, [this] { return stop || !tasks.empty(); });
                        if (stop && tasks.empty()) return; // shutdown
                        task = std::move(tasks.front());
                        tasks.pop();
                    }
                    task(); // execute outside lock
                }
            });
        }
    }

    void enqueue(std::function<void()> task) {
        {
            std::lock_guard<std::mutex> lock(mtx);
            if (stop) throw std::runtime_error("Pool is stopped");
            tasks.push(std::move(task));
        }
        cv.notify_one();
    }

    ~ThreadPool() {
        {
            std::lock_guard<std::mutex> lock(mtx);
            stop = true;
        }
        cv.notify_all(); // wake all workers to let them exit
        for (auto& worker : workers)
            worker.join();
    }
};

int main() {
    ThreadPool pool(4);

    for (int i = 0; i < 8; ++i) {
        pool.enqueue([i] {
            std::cout << "Task " << i << " running on thread "
                      << std::this_thread::get_id() << "\n";
        });
    }
    // Pool destructor waits for all tasks to complete
}
```

## ⚡ Quick Reference: C++ Concurrency API

```cpp
// --- Thread ---
std::thread t(func, arg1, arg2);
t.join();      // wait for t to finish
t.detach();    // let t run independently

// --- Mutex ---
std::mutex m;
m.lock();   m.unlock();
std::lock_guard<std::mutex>  lg(m);          // RAII, no early unlock
std::unique_lock<std::mutex> ul(m);          // RAII, can unlock early, needed for CV

// --- Lock multiple mutexes atomically (avoids deadlock) ---
std::lock(m1, m2);
std::lock_guard<std::mutex> g1(m1, std::adopt_lock);
std::lock_guard<std::mutex> g2(m2, std::adopt_lock);

// --- Condition Variable ---
std::condition_variable cv;
cv.wait(unique_lock, predicate);  // atomically unlock + sleep; recheck predicate on wake
cv.notify_one();                  // wake one waiter
cv.notify_all();                  // wake all waiters

// --- Atomic (lock-free for simple types) ---
std::atomic<int> counter{0};
counter++;          // atomic increment
counter.load();     // atomic read
counter.store(5);   // atomic write
counter.compare_exchange_strong(expected, desired); // CAS

// --- Call once ---
std::once_flag flag;
std::call_once(flag, init_function);  // init_function called exactly once, thread-safe

// --- Future / Promise (async results) ---
std::promise<int> p;
std::future<int>  f = p.get_future();
std::thread t([&p]{ p.set_value(42); });
std::cout << f.get(); // blocks until value is set
t.join();
```
